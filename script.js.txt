const state = {
  current: 0,
  scores: {
    crm: 0,
    automation: 0,
    transaction: 0,
    marketing: 0
  },
  budget: ""
};

const elements = {
  startScreen: document.getElementById("start-screen"),
  questionScreen: document.getElementById("question-screen"),
  resultScreen: document.getElementById("result-screen"),
  startBtn: document.getElementById("start-btn"),
  restartBtn: document.getElementById("restart-btn"),
  questionTitle: document.getElementById("question-title"),
  answers: document.getElementById("answers"),
  progress: document.getElementById("progress"),
  result: document.getElementById("result"),
  year: document.getElementById("year")
};

if (elements.year) {
  elements.year.textContent = new Date().getFullYear();
}

elements.startBtn.addEventListener("click", startQuiz);
elements.restartBtn.addEventListener("click", restartQuiz);

function startQuiz() {
  state.current = 0;
  state.scores = {
    crm: 0,
    automation: 0,
    transaction: 0,
    marketing: 0
  };
  state.budget = "";

  elements.startScreen.hidden = true;
  elements.resultScreen.hidden = true;
  elements.questionScreen.hidden = false;

  showQuestion();
}

function restartQuiz() {
  startQuiz();
}

function showQuestion() {
  const question = questions[state.current];

  elements.progress.textContent = `Question ${state.current + 1} of ${
    questions.length
  }`;

  elements.questionTitle.textContent = question.title;
  elements.answers.innerHTML = "";

  question.answers.forEach((answer) => {
    const button = document.createElement("button");
    button.type = "button";
    button.className = "answer";
    button.textContent = answer.label;

    button.addEventListener("click", () => {
      selectAnswer(answer);
    });

    elements.answers.appendChild(button);
  });
}

function selectAnswer(answer) {
  if (state.current === questions.length - 1) {
    state.budget = answer.label;
  }

  Object.entries(answer.scores || {}).forEach(([category, points]) => {
    state.scores[category] = (state.scores[category] || 0) + points;
  });

  state.current += 1;

  if (state.current < questions.length) {
    showQuestion();
  } else {
    showResult();
  }
}

function getWinner() {
  return Object.entries(state.scores).sort((a, b) => b[1] - a[1])[0][0];
}

function showResult() {
  const winner = getWinner();
  const result = results[winner];
  const budgetNote = budgetNotes[state.budget] || "";

  elements.result.innerHTML = "";

  const title = document.createElement("h3");
  title.textContent = result.title;

  const summary = document.createElement("p");
  summary.textContent = result.summary;

  const budget = document.createElement("p");
  budget.className = "budget-note";
  budget.textContent = budgetNote;

  const toolHeading = document.createElement("h4");
  toolHeading.textContent = "Examples to research";

  const toolList = document.createElement("ul");

  result.tools.forEach((tool) => {
    const li = document.createElement("li");

    const strong = document.createElement("strong");
    strong.textContent = tool.name;

    const span = document.createElement("span");
    span.textContent = ` - ${tool.why}`;

    li.appendChild(strong);
    li.appendChild(span);
    toolList.appendChild(li);
  });

  const stepsHeading = document.createElement("h4");
  stepsHeading.textContent = "Next steps";

  const stepsList = document.createElement("ol");

  result.nextSteps.forEach((step) => {
    const li = document.createElement("li");
    li.textContent = step;
    stepsList.appendChild(li);
  });

  const adPlaceholder = document.createElement("div");
  adPlaceholder.className = "ad-slot result-ad";
  adPlaceholder.textContent = "Advertisement placeholder";

  elements.result.appendChild(title);
  elements.result.appendChild(summary);
  elements.result.appendChild(budget);
  elements.result.appendChild(toolHeading);
  elements.result.appendChild(toolList);
  elements.result.appendChild(stepsHeading);
  elements.result.appendChild(stepsList);
  elements.result.appendChild(adPlaceholder);

  elements.questionScreen.hidden = true;
  elements.resultScreen.hidden = false;
}
